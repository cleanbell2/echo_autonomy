# vocab_tracker.py (SRS 업그레이드 버전)
from __future__ import annotations
from pathlib import Path
import json
from typing import List, Dict, Any, Optional, Set
from dataclasses import dataclass, asdict
import uuid
from collections import defaultdict
from datetime import datetime, timedelta

@dataclass
class VocabEntry:
    id: str
    word: str
    context: str
    definition: str
    tag: str = ""
    timestamp: str = None
    next_review: str = None      # 🚀 다음 검토일
    interval_days: int = 1       # 🚀 반복 간격
    ease_factor: float = 2.5     # 🚀 난이도 조절 (SM2)
    reviews: int = 0             # 🚀 검토 횟수

    def __post_init__(self):
        if self.timestamp is None:
            self.timestamp = datetime.now().isoformat()
        if self.next_review is None:
            self.next_review = self._next_due_date().isoformat()

    def _next_due_date(self) -> datetime:
        return datetime.now() + timedelta(days=self.interval_days)

    def update_review(self, quality: int) -> None:  # quality: 0=잘못, 3=완벽, 4=쉬움
        """SM2 기반 다음 검토 스케줄링"""
        self.reviews += 1
        
        if quality < 3:  # 어려움
            self.interval_days = 1
        else:
            if self.reviews == 1:
                self.interval_days = 1
            elif self.reviews == 2:
                self.interval_days = 6
            else:
                self.interval_days = max(1, int(self.interval_days * self.ease_factor))
        
        self.ease_factor = max(1.3, self.ease_factor + (0.1 - (5-quality) * (0.08 + (5-quality)*0.02)))
        self.next_review = (datetime.now() + timedelta(days=self.interval_days)).isoformat()

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

class VocabTracker:
    # 기존 코드 동일 + 새로운 메서드들만 추가
    def __init__(self, journal_path: Path, reviewed_path: Path):
        self.journal_path = journal_path
        self.reviewed_path = reviewed_path
        self._reviewed_set: Optional[Set[str]] = None
        self.entries: List[VocabEntry] = []

    def mark_reviewed(self, entry_ids: List[str], quality: int = 3):
        """SRS 기반 검토 (quality 점수 입력) 및 리뷰 이벤트 append-only 기록"""
        entries = self._jsonl_iter(self.journal_path)
        updated = []
        for entry in entries:
            if entry["id"] in entry_ids:
                entry_obj = VocabEntry(**entry)
                entry_obj.update_review(quality)
                entry["next_review"] = entry_obj.next_review
                entry["interval_days"] = entry_obj.interval_days
                entry["ease_factor"] = entry_obj.ease_factor
                entry["reviews"] = entry_obj.reviews
                updated.append(entry["id"])
        self._save_jsonl(self.journal_path, entries)
        self._append_review_events(updated, quality)

    def due_today(self, max_words: int = 20) -> List[Dict[str, Any]]:
        """오늘 검토해야 할 단어만 반환 (SRS 핵심!)"""
        today = datetime.now().date().isoformat()
        entries = self._jsonl_iter(self.journal_path)
        
        due_entries = []
        for e in entries:
            if e.get("next_review", "").startswith(today):
                due_entries.append(e)
        
        # 난이도별 우선순위 (어려운 단어 우선)
        due_entries.sort(key=lambda x: x.get("ease_factor", 2.5))
        return due_entries[:max_words]

    def _append_review_events(self, entry_ids: List[str], quality: int):
        """reviewed_ids.jsonl에 {id, ts, q} append-only 기록"""
        with open(self.reviewed_path, "a", encoding="utf-8") as f:
            for eid in entry_ids:
                event = {"id": eid, "ts": datetime.now().isoformat(), "q": quality}
                f.write(json.dumps(event, ensure_ascii=False) + "\n")
    def _get_reviewed_id_set(self) -> Set[str]:
        """reviewed_ids.jsonl에서 이미 복습된 id 집합 반환"""
        reviewed_ids = set()
        try:
            with open(self.reviewed_path, "r", encoding="utf-8") as f:
                for line in f:
                    try:
                        row = json.loads(line)
                        if "id" in row:
                            reviewed_ids.add(row["id"])
                    except Exception:
                        continue
        except FileNotFoundError:
            pass
        return reviewed_ids
    def export_new_entries(self, max_words: int = 20) -> List[Dict[str, Any]]:
        """아직 한 번도 복습되지 않은 신규 단어만 반환"""
        reviewed_ids = self._get_reviewed_id_set()
        entries = self._jsonl_iter(self.journal_path)
        new_entries = [e for e in entries if e["id"] not in reviewed_ids]
        return new_entries[:max_words]
