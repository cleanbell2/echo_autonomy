html<!-- 초간단 HTML 버전 -->
<!DOCTYPE html>
<html>
<head>
    <title>AI Safety Scanner</title>
    <style>
        body { 
            font-family: Arial; 
            max-width: 600px; 
            margin: 50px auto; 
            padding: 20px;
        }
        .danger { color: red; }
        .safe { color: green; }
    </style>
</head>
<body>
    <h1>🛡️ AI Safety Scanner</h1>
    <p>당신의 AI 질문이 안전한지 확인해보세요</p>
    
    <textarea id="question" placeholder="AI에게 물어볼 질문을 입력하세요" 
              style="width: 100%; height: 100px;"></textarea>
    
    <button onclick="checkSafety()">안전도 분석</button>
    
    <div id="result"></div>
    
    <script>
        function checkSafety() {
            const question = document.getElementById('question').value;
            
            // 초간단 위험 키워드 체크
            const dangerous = ['핵', '폭탄', '해킹', '불법'];
            const isDangerous = dangerous.some(word => question.includes(word));
            
            const result = document.getElementById('result');
            
            if (isDangerous) {
                result.innerHTML = `
                    <div class="danger">
                        <h2>🔴 위험 감지!</h2>
                        <p>위험도: 80%</p>
                        <p>이 질문은 위험할 수 있습니다</p>
                    </div>
                `;
            } else {
                result.innerHTML = `
                    <div class="safe">
                        <h2>✅ 안전합니다!</h2>
                        <p>위험도: 15%</p>
                        <p>안전한 질문입니다</p>
                    </div>
                `;
            }
        }
    </script>
</body>
</html>
```
